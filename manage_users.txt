// Helper to validate type and set table/column
function get_table_info($type) {
    if ($type === 'student') {
        return ['table' => 'student_info', 'id_col' => 'sid', 'name_col' => 'stu_fname', 'lname_col' => 'stu_lname', 'email_col' => 'stu_email'];
    } elseif ($type === 'employer') {
        return ['table' => 'employer_info', 'id_col' => 'eid', 'company_col' => 'com_name', 'email_col' => 'emp_email'];
    }
    return false;
}

// Delete user: uses POST, updates DOM on success
async function deleteUser(id, type){
    if(!confirm('Are you sure you want to delete this user?')) return;
    try {
        let form = new URLSearchParams();
        form.append('delete_user', '1');
        form.append('id', id);
        form.append('type', type);

        let res = await fetch('manage_users.php', {
            method: 'POST',
            headers: {'Content-Type': 'application/x-www-form-urlencoded'},
            body: form.toString()
        });
        let data = await res.json();
        if(data.status === 'success'){
            let row = document.getElementById(type + '_' + id);
            if(row) row.remove();
        } else {
            alert('Error deleting user: ' + (data.message || 'Unknown error'));
        }
    } catch (err) {
        console.error(err);
        alert('Request failed');
    }
}

// Block/Unblock user: uses GET, updates status cell and buttons on success
async function actionUser(id, type, action){
    try {
        let url = 'manage_users.php?action=' + encodeURIComponent(action) + '&type=' + encodeURIComponent(type) + '&id=' + encodeURIComponent(id);
        let res = await fetch(url, { method: 'GET' });
        let data = await res.json();
        if(data.status === 'success'){
            // update status text
            let row = document.getElementById(type + '_' + id);
            if(!row) return;
            // find the status cell (assumed 4th column for students/employers as in markup)
            let statusCell = row.querySelector('.status-cell');
            if(statusCell) statusCell.textContent = data.new_status.charAt(0).toUpperCase() + data.new_status.slice(1);

            // replace action buttons with correct ids and onclicks
            let actionsCell = row.querySelector('.actions-cell');
            if(actionsCell){
                actionsCell.innerHTML = ''; // clear
                if(data.new_status === 'blocked'){
                    // show Unblock + Delete
                    let unblockBtn = document.createElement('button');
                    unblockBtn.className = 'btn btn-unblock';
                    unblockBtn.id = 'unblock_' + type + '_' + id;
                    unblockBtn.textContent = 'Unblock';
                    unblockBtn.onclick = function(){ actionUser(id, type, 'unblock'); };

                    let delBtn = document.createElement('button');
                    delBtn.className = 'btn btn-delete';
                    delBtn.textContent = 'Delete';
                    delBtn.onclick = function(){ deleteUser(id, type); };

                    actionsCell.appendChild(unblockBtn);
                    actionsCell.appendChild(delBtn);
                } else {
                    // show Block + Delete
                    let blockBtn = document.createElement('button');
                    blockBtn.className = 'btn btn-block';
                    blockBtn.id = 'block_' + type + '_' + id;
                    blockBtn.textContent = 'Block';
                    blockBtn.onclick = function(){ actionUser(id, type, 'block'); };

                    let delBtn = document.createElement('button');
                    delBtn.className = 'btn btn-delete';
                    delBtn.textContent = 'Delete';
                    delBtn.onclick = function(){ deleteUser(id, type); };

                    actionsCell.appendChild(blockBtn);
                    actionsCell.appendChild(delBtn);
                }
            }
        } else {
            alert('Action failed: ' + (data.message || 'Unknown error'));
        }
    } catch(err){
        console.error(err);
        alert('Request failed');
    }
}
